}

async function loadAdminContent(){
if(!requireAdmin())return;

$("adminContent").textContent="লোড হচ্ছে...";

try{
const [hs,ps]=await Promise.all([
getDocs(collection(db,"hadiths")),
getDocs(collection(db,"posts"))
]);

$("adminContent").innerHTML="";

hs.forEach(d=>{
const div=document.createElement("div");
div.className="item";
div.textContent="হাদিস: "+d.data().book+" — "+d.data().number;

const b=document.createElement("button");
b.className="danger";
b.textContent="মুছুন";
b.onclick=()=>deleteItem("hadiths",d.id);

div.appendChild(b);
$("adminContent").appendChild(div);
});

ps.forEach(d=>{
const div=document.createElement("div");
div.className="item";
div.textContent="পোস্ট: "+d.data().title;

const b=document.createElement("button");
b.className="danger";
b.textContent="মুছুন";
b.onclick=()=>deleteItem("posts",d.id);

div.appendChild(b);
$("adminContent").appendChild(div);
});

if(hs.empty&&ps.empty){
$("adminContent").textContent="এখনো কোনো কনটেন্ট নেই।";
}

}catch(e){
$("adminContent").textContent="লোড সমস্যা: "+e.message;
}
}

async function deleteItem(type,id){
if(!requireAdmin()||!confirm("আপনি কি নিশ্চিতভাবে মুছতে চান?"))return;

try{
await deleteDoc(doc(db,type,id));
loadAdminContent();
loadPublicData();

}catch(e){
alert("মুছতে সমস্যা: "+e.message);
}
}

loadPublicData();
